\newcommand{\padlock}[3][1]{%
  \begin{scope}[shift={(#2,#3)},scale=#1]

\newcommand{\shield}[4][1]{% x y colour
  \begin{scope}[shift={(#2,#3)},scale=#1]
  \fill[#4] (0,0.42) .. controls (0.2,0.34) .. (0.32,0.34) -- (0.32,0.04) .. controls (0.32,-0.23) .. (0,-0.38)

  \end{scope}}
